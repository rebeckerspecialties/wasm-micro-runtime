cmake_minimum_required(VERSION 3.14)
project(core_linking C ASM)
get_filename_component(WAMR_ROOT_DIR "../.." ABSOLUTE)
string(TOLOWER ${CMAKE_HOST_SYSTEM_NAME} WAMR_BUILD_PLATFORM)
if(APPLE)
  add_definitions(-DBH_PLATFORM_DARWIN)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)")
  set(WAMR_BUILD_TARGET AARCH64)
else()
  set(WAMR_BUILD_TARGET X86_64)
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_GC 0)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_THREAD_MGR 1)
set(WAMR_BUILD_ALLOCATION_QUOTA 1)
if(NOT DEFINED WAMR_BUILD_C_API_CORE_LINKING)
  set(WAMR_BUILD_C_API_CORE_LINKING 1)
endif()
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_BUILD_MINI_LOADER 0)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(vmlib STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_link_libraries(vmlib PUBLIC m pthread)
find_program(WAT2WASM wat2wasm REQUIRED)
foreach(fixture memory-owner memory-user start-import table-owner table-fixup table-trap start-loop start-trap)
  add_custom_command(OUTPUT ${fixture}.wasm
    COMMAND ${WAT2WASM} ${CMAKE_CURRENT_SOURCE_DIR}/${fixture}.wat -o ${fixture}.wasm
    DEPENDS ${fixture}.wat VERBATIM)
  list(APPEND FIXTURES ${fixture}.wasm)
endforeach()
add_custom_target(fixtures ALL DEPENDS ${FIXTURES})
add_executable(core_linking_test linking_test.c)
target_link_libraries(core_linking_test PRIVATE vmlib)
add_dependencies(core_linking_test fixtures)
enable_testing()
add_test(NAME core_linking COMMAND core_linking_test)
add_executable(instance_policy_test instance_policy_test.c)
target_link_libraries(instance_policy_test PRIVATE vmlib)
add_dependencies(instance_policy_test fixtures)
add_test(NAME instance_policy COMMAND instance_policy_test)
set_tests_properties(instance_policy PROPERTIES TIMEOUT 10)
